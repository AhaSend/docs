These AhaSend CLI examples show common email sending and management tasks, including sandbox tests and automation.

Before running the examples, [log in with a full API key](/cli/authentication), use your verified sender domain, and replace `YOUR_RESOURCE_ID` with a resource UUID from your test account. Create any named template, recipient, attachment, or JSON files first. Shell examples use Bash and Python 3 for portable future dates; examples that read JSON with `jq` need jq installed. Sends use sandbox mode.

## Transactional Emails

### Order Confirmation

```bash
# Single order confirmation
ahasend messages send --sandbox \
  --from orders@shop.com \
  --to customer@example.com \
  --subject "Order #12345 Confirmed" \
  --html-template order-confirmation.html \
  --global-substitutions order-data.json \
  --attach invoice.pdf \
  --idempotency-key "order-12345"
```

**order-data.json:**
```json
{
  "order_id": "12345",
  "customer_name": "John Doe",
  "total": "$149.99",
  "items": [
    {"name": "Product A", "quantity": 2, "price": "$49.99"},
    {"name": "Product B", "quantity": 1, "price": "$50.01"}
  ],
  "delivery_date": "<delivery date>"
}
```

## Lifecycle Notifications

### Account Activity Digest

```bash
#!/bin/bash
# send-newsletter.sh

# Prepare data
TEMPLATE="account-digest.html"
RECIPIENTS="account-users.csv"
SUBJECT="Your Account Activity Digest"

# Send newsletter with tracking
ahasend messages send --sandbox \
  --from newsletter@company.com \
  --recipients "$RECIPIENTS" \
  --subject "$SUBJECT" \
  --html-template "$TEMPLATE" \
  --track-opens=true \
  --track-clicks=true \
  --tags account-digest,transactional \
  --max-concurrency 5 \
  --progress \
  --show-metrics

# Check delivery stats after 1 hour
# Check recent statistics after the sandbox send.
ahasend stats deliverability \
  --from-time 7d \
  --group-by hour
```

## Notification Systems

### Real-Time Alerts

```bash
#!/bin/bash
# send-alert.sh

ALERT_TYPE=${1:-info}
ALERT_MESSAGE=${2:-"Docs alert test"}
RECIPIENTS=${3:-test@example.com}

case $ALERT_TYPE in
  critical)
    SUBJECT="🚨 CRITICAL ALERT"
    PRIORITY="high"
    ;;
  warning)
    SUBJECT="⚠️ Warning"
    PRIORITY="normal"
    ;;
  info)
    SUBJECT="ℹ️ Information"
    PRIORITY="low"
    ;;
  *) echo "Use critical, warning, or info"; exit 1 ;;
esac

ahasend messages send --sandbox \
  --from alerts@monitoring.com \
  --to "$RECIPIENTS" \
  --subject "$SUBJECT: $ALERT_MESSAGE" \
  --text "$(printf '%s\n\nTime: %s\nSystem: Test' "$ALERT_MESSAGE" "$(date)")" \
  --header "X-Priority: $PRIORITY" \
  --tags "alert,${ALERT_TYPE},automated"
```

## Webhook Integration

### Process Webhook Events

```bash
# Run your local, signature-verifying receiver first; see the webhook guide.
ahasend webhooks listen \
  --forward-to http://localhost:8080/webhook \
  --events message.bounced,message.failed
```

The receiver processes `payload.type` and `payload.data.recipient`. CLI listener output is for people, not a line-delimited JSON stream. See [webhook verification and processing](/integrations/webhooks). AhaSend already creates cause-dependent automatic suppressions; see [suppression durations](/bounces#suppression-duration).

### Webhook Testing Pipeline

```bash
#!/bin/bash
# test-webhook-integration.sh
set -euo pipefail

# 1. Create test webhook
WEBHOOK_ID=$(ahasend webhooks create --scope global --non-interactive --disabled \
  --url https://example.com/test-webhook \
  --all-events \
  --name "Integration test webhook" \
  --output json | jq -er '.id')

cleanup() {
  ahasend webhooks update "$WEBHOOK_ID" --disable || true
  if [ -n "${LISTENER_PID:-}" ]; then kill "$LISTENER_PID" 2>/dev/null || true; fi
  ahasend webhooks delete "$WEBHOOK_ID" --force
}
trap cleanup EXIT

echo "Created webhook: $WEBHOOK_ID"

# 2. Start listener with forwarding (a local handler must already be running)
ahasend webhooks listen \
  --webhook-id "$WEBHOOK_ID" \
  --forward-to http://localhost:3000/test-webhook \
  --verbose &
LISTENER_PID=$!

# 3. Trigger test events
sleep 2
# Enable only after the listener connects; disabled webhooks emit no events.
ahasend webhooks update "$WEBHOOK_ID" --enable
ahasend webhooks trigger "$WEBHOOK_ID" --all-events

# 4. Send test email
ahasend messages send \
  --from test@example.com \
  --to test@example.com \
  --subject "Webhook Test" \
  --text "Testing webhook integration" \
  --sandbox

# 5. Wait and cleanup
sleep 5
echo "Integration test complete"
```

## Automation Scripts

### CI/CD Email Notifications

```yaml
# .github/workflows/deploy-notification.yml
name: Deployment Notification

on:
  deployment_status:

jobs:
  notify:
    if: github.event.deployment_status.state == 'success'
    runs-on: ubuntu-latest
    steps:
      - name: Install AhaSend CLI
        run: |
          wget https://github.com/AhaSend/ahasend-cli/releases/latest/download/ahasend-linux-amd64
          chmod +x ahasend-linux-amd64
          sudo mv ahasend-linux-amd64 /usr/local/bin/ahasend

      - name: Send deployment notification
        env:
          AHASEND_API_KEY: ${{ secrets.AHASEND_API_KEY }}
          AHASEND_ACCOUNT_ID: ${{ secrets.AHASEND_ACCOUNT_ID }}
        run: |
          ahasend --api-key "$AHASEND_API_KEY" --account-id "$AHASEND_ACCOUNT_ID" messages send --sandbox \
            --from ci@company.com \
            --to team@company.com \
            --subject "✅ Deployment Successful: ${{ github.event.deployment.environment }}" \
            --html "<h2>Deployment Complete</h2>
                    <p>Environment: ${{ github.event.deployment.environment }}</p>
                    <p>Commit: ${{ github.sha }}</p>
                    <p>Deployed by: ${{ github.actor }}</p>
                    <p>Time: $(date)</p>" \
            --tags "deployment,ci,notification"
```

### Database Backup Notifications

```bash
#!/bin/bash
# backup-notification.sh

# Set BACKUP_SOURCE to your database dump. This small fixture makes the test runnable.
printf '%s
' 'SELECT 1;' > docs-backup-source.sql
BACKUP_SOURCE=docs-backup-source.sql
BACKUP_FILE="backup-$(date +%Y%m%d-%H%M%S).sql.gz"
gzip -c "$BACKUP_SOURCE" > "$BACKUP_FILE"
BACKUP_STATUS=$?
BACKUP_SIZE=$(wc -c < "$BACKUP_FILE")

if [ $BACKUP_STATUS -eq 0 ]; then
  SUBJECT="✅ Backup Successful"
  MESSAGE=$(printf 'Backup completed.\nFile: %s\nBytes: %s' "$BACKUP_FILE" "$BACKUP_SIZE")
  TAGS="backup,success"
else
  SUBJECT="❌ Backup Failed"
  MESSAGE="Database backup failed with error code: $BACKUP_STATUS"
  TAGS="backup,failure,alert"
fi

# Send notification
ahasend messages send --sandbox \
  --from backup@database.com \
  --to admin@company.com \
  --subject "$SUBJECT - $(date +%Y-%m-%d)" \
  --text "$MESSAGE" \
  --tags "$TAGS"
```

## Performance Testing

### Load Testing Script

```bash
#!/bin/bash
# load-test.sh

# Configuration
TOTAL_EMAILS=${TOTAL_EMAILS:-10}
CONCURRENCY=10

echo "Starting load test: $TOTAL_EMAILS emails"

# Generate test recipients
echo "email,name,test_id" > test-recipients.csv
for i in $(seq 1 $TOTAL_EMAILS); do
  echo "test${i}@example.com,Test User ${i},TEST-${i}" >> test-recipients.csv
done

# Run load test
START_TIME=$(date +%s)

ahasend messages send \
  --from loadtest@example.com \
  --recipients test-recipients.csv \
  --subject "Load Test {{test_id}}" \
  --text "This is a load test message for {{name}}" \
  --sandbox \
  --max-concurrency $CONCURRENCY \
  --progress \
  --show-metrics \
  --output json > load-test-results.json

END_TIME=$(date +%s)
DURATION=$((END_TIME - START_TIME))

# The CLI does not expose successful_jobs, failed_jobs, or emails_per_sec.
# Inspect the returned JSON and elapsed time without assuming those fields.
cat load-test-results.json
echo "Elapsed: ${DURATION}s; requested recipients: $TOTAL_EMAILS"

# Cleanup
rm test-recipients.csv
```

## Multi-Environment Management

### Environment-Specific Sending

```bash
#!/bin/bash
# deploy-emails.sh

ENVIRONMENT=$1

case $ENVIRONMENT in
  production)
    PROFILE="production"
    FROM="noreply@company.com"
    RECIPIENTS="all-users.csv"
    ;;
  staging)
    PROFILE="staging"
    FROM="test@staging.company.com"
    RECIPIENTS="test-users.csv"
    ;;
  development)
    PROFILE="development"
    FROM="dev@dev.company.com"
    RECIPIENTS="dev-team.csv"
    ;;
  *)
    echo "Unknown environment: $ENVIRONMENT"
    exit 1
    ;;
esac

# Send with environment-specific profile
ahasend messages send --sandbox \
  --profile "$PROFILE" \
  --from "$FROM" \
  --recipients "$RECIPIENTS" \
  --subject "[$ENVIRONMENT] System Update" \
  --html-template update-notification.html \
  --tags "update,${ENVIRONMENT}"

echo "Emails sent to $ENVIRONMENT environment"
```
